const PASSWORD = "2008";

let inputPassword = "";


/* =====================================================
   NHẬP SỐ
===================================================== */

function press(number) {

    if (
        inputPassword.length >= 4
    ) {

        return;
    }


    inputPassword += number;

    updateDots();


    if (
        inputPassword.length ===
        PASSWORD.length
    ) {

        setTimeout(() => {

            if (
                inputPassword ===
                PASSWORD
            ) {

                unlock();

            } else {

                wrongPassword();

            }

        }, 300);
    }
}


/* =====================================================
   DẤU CHẤM
===================================================== */

function updateDots() {

    for (
        let i = 1;
        i <= 4;
        i++
    ) {

        const dot =
            document.getElementById(
                "dot" + i
            );


        if (
            i <= inputPassword.length
        ) {

            dot.classList.add(
                "active"
            );

        } else {

            dot.classList.remove(
                "active"
            );
        }
    }
}


/* =====================================================
   XÓA
===================================================== */

function removeNumber() {

    inputPassword =
        inputPassword.slice(
            0,
            -1
        );

    updateDots();
}

   SAI MẬT KHẨU
===================================================== */

    const dots =

}

/* =====================================================
   GÕ TỪNG CHỮ
===================================================== */

function typeText(
    element,
    text,
    speed
) {

    return new Promise(
        resolve => {

            let index = 0;


            const timer =
                setInterval(() => {


                    element.textContent +=
                        text[index];


                    index++;


                    if (
                        index >=
                        text.length
                    ) {

                        clearInterval(
                            timer
                        );

                        resolve();
                    }


                }, speed);

        }
    );
}

function createFirework(
    x,
    y
) {

    const colors = [

        "#ff1744",
        "#ffea00",
        "#00e5ff",
        "#76ff03",
        "#d500f9",
        "#ff9100",
        "#ffffff",
        "#00ff88"

    ];


    const color =
        colors[
            Math.floor(
                Math.random() *
                colors.length
            )
        ];


    const count = 90;


    for (
        let i = 0;
        i < count;
        i++
    ) {

        const angle =
            Math.random() *
            Math.PI * 2;


        const speed =
            2 +
            Math.random() * 5;


        particles.push({

            x: x,

            y: y,

            vx:
                Math.cos(angle)
                * speed,

            vy:
                Math.sin(angle)
                * speed,

            life:
                80 +
                Math.random() * 40,

            color:
                color,

            size:
                1.5 +
                Math.random() * 2

        });
    }
}


/* =====================================================
   VẼ PHÁO HOA
===================================================== */

function animateFireworks() {

    if (!fireworkRunning) {

        return;
    }


    ctx.fillStyle =
        "rgba(2,8,23,.20)";


    ctx.fillRect(
        0,
        0,
        canvas.width,
        canvas.height
    );


    /* Tạo pháo hoa ngẫu nhiên */

    if (
        Math.random() <
        0.045
    ) {

        createFirework(

            80 +
            Math.random() *
            (canvas.width - 160),

            80 +
            Math.random() *
            (canvas.height * .55)

        );
    }


    /* Hạt pháo hoa */

    for (
        let i = particles.length - 1;
        i >= 0;
        i--
    ) {

        const p =
            particles[i];


        p.x += p.vx;

        p.y += p.vy;

        p.vy += .035;

        p.vx *= .99;

        p.vy *= .99;

        p.life--;


        ctx.globalAlpha =
            Math.max(
                p.life / 120,
                0
            );


        ctx.beginPath();

        ctx.arc(
            p.x,
            p.y,
            p.size,
            0,
            Math.PI * 2
        );

        ctx.fillStyle =
            p.color;

        ctx.fill();


        if (
            p.life <= 0
        ) {

            particles.splice(
                i,
                1
            );
        }
    }


    ctx.globalAlpha = 1;


    requestAnimationFrame(
        animateFireworks
    );
}

/* =====================================================
   PHÁO HOA HÌNH TRÁI TIM
===================================================== */

function createHeartFirework() {

    const cx =
        100 +
        Math.random() *
        (canvas.width - 200);


    const cy =
        100 +
        Math.random() *
        (canvas.height * .45);


    const colors = [

        "#ff1744",
        "#ff4081",
        "#e040fb",
        "#ffea00",
        "#00e5ff",
        "#76ff03"

    ];


    const color =
        colors[
            Math.floor(
                Math.random() *
                colors.length
            )
        ];


    for (
        let i = 0;
        i < 120;
        i++
    ) {

        const t =
            Math.PI * 2 *
            i / 120;


        /*
           Phương trình trái tim
        */

        const hx =
            16 *
            Math.pow(
                Math.sin(t),
                3
            );


        const hy =
            -(
                13 *
                Math.cos(t)
                - 5 *
                Math.cos(2*t)
                - 2 *
                Math.cos(3*t)
                - Math.cos(4*t)
            );


        const scale =
            0.22 +
            Math.random() * .12;


        particles.push({

            x: cx,

            y: cy,

            vx:
                hx * scale,

            vy:
                hy * scale,

            life:
                100 +
                Math.random() * 40,

            color:
                color,

            size:
                2 +
                Math.random() * 2

        });
    }
}
